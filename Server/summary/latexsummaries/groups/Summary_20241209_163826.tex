\subsection{Fairness in Machine Learning}%
\label{subsec:FairnessinMachineLearning}%
\begin{itemize}[leftmargin=*]%
\item \href{https://purdue.brightspace.com/d2l/common/assets/pdfjs-d2l-dist/1.0.14-legacy/web/viewer.html?file=\%2Fcontent\%2Fenforced\%2F1095467-wl.202510.CS.24300.001\%2F11L2-svd-pca-recommender.pdf\%3Fou\%3D1095467\&lang=en-us\&container=d2l-fileviewer-rendered-pdf\&fullscreen=d2l-fileviewer-rendered-pdf-dialog\&height=667#page=44}{\textbf{Modeling Systematic Biases in Recommender Systems}}: $r_{ij} \approx \mu + b_i + b_j + u_i \cdot v_j$, where $r_{ij}$ is the rating, $\mu$ is the overall mean rating, $b_i$ is the user bias, $b_j$ is the item bias, and $u_i \cdot v_j$ represents user-item interactions.%
\item \href{https://purdue.brightspace.com/d2l/common/assets/pdfjs-d2l-dist/1.0.14-legacy/web/viewer.html?file=\%2Fcontent\%2Fenforced\%2F1095467-wl.202510.CS.24300.001\%2F16L2-ethics.pdf\%3Fou\%3D1095467\&lang=en-us\&container=d2l-fileviewer-rendered-pdf\&fullscreen=d2l-fileviewer-rendered-pdf-dialog\&height=667#page=25}{\textbf{Fairness through Blindness}}: Ignore all irrelevant/protected attributes. However, it is possible to predict these attributes with high accuracy from other data.%
\item \href{https://purdue.brightspace.com/d2l/common/assets/pdfjs-d2l-dist/1.0.14-legacy/web/viewer.html?file=\%2Fcontent\%2Fenforced\%2F1095467-wl.202510.CS.24300.001\%2F16L2-ethics.pdf\%3Fou\%3D1095467\&lang=en-us\&container=d2l-fileviewer-rendered-pdf\&fullscreen=d2l-fileviewer-rendered-pdf-dialog\&height=667#page=26}{\textbf{Group Fairness}}: Equalize two groups $S$, $T$ at the level of outcomes. $P(\text{outcome } o | S) = P(\text{outcome } o | T)$. This is not a strong enough notion of fairness.%
\item \href{https://purdue.brightspace.com/d2l/common/assets/pdfjs-d2l-dist/1.0.14-legacy/web/viewer.html?file=\%2Fcontent\%2Fenforced\%2F1095467-wl.202510.CS.24300.001\%2F16L2-ethics.pdf\%3Fou\%3D1095467\&lang=en-us\&container=d2l-fileviewer-rendered-pdf\&fullscreen=d2l-fileviewer-rendered-pdf-dialog\&height=667#page=27}{\textbf{Individual Fairness}}: Treat similar individuals similarly. Similar individuals should be treated similarly for the purpose of the classification task and should be similarly distributed over outcomes.%
\end{itemize}

\subsection{Data Analysis \& Preprocessing}%
\label{subsec:DataAnalysisPreprocessing}%
\begin{itemize}[leftmargin=*]%
\item \href{https://purdue.brightspace.com/d2l/common/assets/pdfjs-d2l-dist/1.0.14-legacy/web/viewer.html?file=\%2Fcontent\%2Fenforced\%2F1095467-wl.202510.CS.24300.001\%2F07L2-Vision.pdf\%3Fou\%3D1095467\&lang=en-us\&container=d2l-fileviewer-rendered-pdf\&fullscreen=d2l-fileviewer-rendered-pdf-dialog\&height=667#page=38}{\textbf{Moving Average}}: Replace each pixel with an average of all the values in its neighborhood%
\item \href{https://purdue.brightspace.com/d2l/common/assets/pdfjs-d2l-dist/1.0.14-legacy/web/viewer.html?file=\%2Fcontent\%2Fenforced\%2F1095467-wl.202510.CS.24300.001\%2F07L2-Vision.pdf\%3Fou\%3D1095467\&lang=en-us\&container=d2l-fileviewer-rendered-pdf\&fullscreen=d2l-fileviewer-rendered-pdf-dialog\&height=667#page=40}{\textbf{Weighted Moving Average}}: Add weights to our moving average%
\item \href{https://purdue.brightspace.com/d2l/common/assets/pdfjs-d2l-dist/1.0.14-legacy/web/viewer.html?file=\%2Fcontent\%2Fenforced\%2F1095467-wl.202510.CS.24300.001\%2F12L2-distance-clusters.pdf\%3Fou\%3D1095467\&lang=en-us\&container=d2l-fileviewer-rendered-pdf\&fullscreen=d2l-fileviewer-rendered-pdf-dialog\&height=667#page=17}{\textbf{Euclidean Distance}}: $d_E(x, y) = \sqrt{\sum_{i=1}^{p} (x_i - y_i)^2}$, where $x$ and $y$ are $p$-dimensional vectors.%
\item \href{https://purdue.brightspace.com/d2l/common/assets/pdfjs-d2l-dist/1.0.14-legacy/web/viewer.html?file=\%2Fcontent\%2Fenforced\%2F1095467-wl.202510.CS.24300.001\%2F12L2-distance-clusters.pdf\%3Fou\%3D1095467\&lang=en-us\&container=d2l-fileviewer-rendered-pdf\&fullscreen=d2l-fileviewer-rendered-pdf-dialog\&height=667#page=19}{\textbf{Hamming Distance}}: Counts the number of differing bits between two binary vectors.%
\item \href{https://purdue.brightspace.com/d2l/common/assets/pdfjs-d2l-dist/1.0.14-legacy/web/viewer.html?file=\%2Fcontent\%2Fenforced\%2F1095467-wl.202510.CS.24300.001\%2F12L2-distance-clusters.pdf\%3Fou\%3D1095467\&lang=en-us\&container=d2l-fileviewer-rendered-pdf\&fullscreen=d2l-fileviewer-rendered-pdf-dialog\&height=667#page=19}{\textbf{Matching Coefficient}}: Hamming distance normalized by the number of attributes/bits.%
\end{itemize}
