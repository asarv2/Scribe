\textbf{Integer Programming}

\begin{equation*}
\max \ \vec{c}^T \vec{x}
\end{equation*}
\begin{equation*}
\text{s.t.} \quad A\vec{x} \le \vec{b}
\end{equation*}
\begin{equation*}
\vec{x} \ge \vec{0}, \quad x_j \in \mathbb{Z}
\end{equation*}

\textit{relaxed problem} $\implies$ $\max \ \vec{c}^T \vec{x}$
\begin{equation*}
\text{s.t.} \quad A\vec{x} \le \vec{b}
\end{equation*}
\begin{equation*}
\vec{x} \ge \vec{0}
\end{equation*}

$Rk$: Solving the problem without the int. constraints and then round up/down the solution does not work because
\begin{enumerate}
    \item rounded Solution might not be feasible
    \item true opt. int. solution might be far from the rounded Solution.
\end{enumerate}
