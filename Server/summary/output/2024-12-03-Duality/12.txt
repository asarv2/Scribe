\textbf{Sensitivity Analysis (what if $c$, $b$ change)}

\vspace{0.5cm}

$(P) \quad \text{max} \quad \mathcal{S} = c^T x$
\vspace{0.5cm}
s.t. $\quad Ax \le b$
\vspace{0.5cm}
$\quad x \ge 0$

\vspace{0.5cm}

$(D) \quad \text{min} \quad \mathcal{S} = b^T y$
\vspace{0.5cm}
s.t. $\quad A^T y \ge c$
\vspace{0.5cm}
$\quad y \ge 0$

\vspace{0.5cm}

$\mathcal{S}(x_B, x_N) = c_B^T \bar{x}_B - (c_N^T - (\bar{B}^T \bar{N})^T c_B) x_N$

$\bar{x}_B = \bar{B}^{-1} b - (\bar{B}^{-1} \bar{N}) x_N$

$\bar{x}_B = \bar{B}^{-1} b$

$(\bar{B}^T \bar{N})^T c_N - c_N \ge 0$

$\mathcal{S} = c^T x^*$
