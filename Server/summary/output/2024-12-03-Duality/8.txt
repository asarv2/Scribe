\begin{align*}
\label{eq:1}
(P) \quad &\text{max} \quad x_1 + x_2 = 10 \\
&\text{s.t.} \quad 2x_1 + x_2 \le 14 \\
& \quad \quad -x_1 + 2x_2 \le 8 \\
& \quad \quad 2x_1 - x_2 \le 10 \\
& \quad \quad x_1, x_2 \ge 0
\end{align*}

\vspace{0.5cm}

\begin{align*}
(D) \quad &\text{min} \quad 14y_1 + 8y_2 + 10y_3 = 10 \\
&\text{s.t.} \quad 2y_1 - y_2 + 2y_3 \ge 1 \\
& \quad \quad y_1 + 2y_2 - y_3 \ge 1 \\
& \quad \quad y_1 - 2y_2 + 2y_3 \ge 0 \\
& \quad \quad y_1, y_2, y_3 \ge 0
\end{align*}

\textbf{Optimal Solution (P)}

$x_1 = 4$, $x_2 = 6$
$z_1 = 0$, $z_2 = 0$, $z_3 = 0$

\textbf{Optimal Solution (D)}

$y_1 = \frac{3}{5}$, $y_2 = \frac{1}{5}$, $y_3 = 0$
$w_1 = 0$, $w_2 = 0$, $w_3 > 0$
