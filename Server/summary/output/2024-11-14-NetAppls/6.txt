\section*{Inequality constraints (Transportation Problem)}

\textbf{Case N1}
minimize $\sum_{i=1}^{2} \sum_{j=3}^{4} c_{ij} x_{ij}$
subject to
\begin{itemize}
    \item $x_{13} + x_{14} \ge -4$
    \item $x_{23} + x_{24} \ge -9$
    \item $x_{13} + x_{23} = 7$
    \item $x_{14} + x_{24} = 2$
    \item $x_{13}, x_{14}, x_{23}, x_{24} \ge 0$
\end{itemize}

\textbf{Case N2}
minimize $\sum_{i=1}^{2} \sum_{j=3}^{4} c_{ij} x_{ij}$
subject to
\begin{itemize}
    \item $x_{13} + x_{21} \ge -4$
    \item $x_{15} + x_{25} \ge -9$
    \item $x_{13} + x_{23} = 7$
    \item $x_{24} + x_{34} = 2$
    \item $x_{15} + x_{25} = 4$
    \item $x_{13}, x_{21}, x_{23}, x_{24}, x_{34}, x_{15}, x_{25} \ge 0$
\end{itemize}
