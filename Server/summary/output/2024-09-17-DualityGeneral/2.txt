\section*{Equality Constraints}

\begin{itemize}
    \item \textbf{Problem:}
    \begin{align*}
        \text{maximize } \zeta &= f_0(x) \\
        x \in \mathbb{R}^d \\
        \text{subject to } g_j(x) &= 0 \quad j=1,\dots,n
    \end{align*}
    \item \textbf{Lagrange Multiplier( $\mu$ ) \& Lagrangian (L)}
    \begin{align*}
        L(x,\mu) &= f_0(x) - \sum_{j=1}^n \mu_j g_j(x)
    \end{align*}
    \begin{itemize}
        \item $x_i, \mu_1, \dots, \mu_m$ are variables.
        \item $j=1,\dots,n$ denotes the number of equality constraints.
    \end{itemize}
\end{itemize}
