\textbf{Duality for General LP}

\begin{itemize}
    \item \textbf{Primal Problem (P)}
    \begin{align*}
        \max \xi(x) &= c^T x \\
        \text{s.t.} \quad Ax &\le b \\
        x &\ge 0
    \end{align*}
    \item \textbf{Dual Problem (D)}
    \begin{align*}
        \min \xi(\lambda) &= \lambda^T b \\
        \text{s.t.} \quad A^T \lambda &\ge c \\
        \lambda &\ge 0
    \end{align*}
    \item \textbf{Lagrangian Function}
    \begin{align*}
        L(x,\lambda) &= c^T x - \lambda^T (Ax - b) = c^T x - \lambda^T Ax + \lambda^T b
    \end{align*}
    \item $\max_x L(x,\lambda) = c^T x - \lambda^T Ax + \lambda^T b$
    \item $\max_x L(x,\lambda) = \begin{cases}
        \infty & \text{if } c^T x - \lambda^T Ax + \lambda^T b \text{ is unbounded} \\
        \lambda^T b & \text{if } c^T x - \lambda^T Ax + \lambda^T b \text{ is bounded}
    \end{cases}$
\end{itemize}

The key point is that the maximum value of $L(x,\lambda)$ is either unbounded or equal to $\lambda^T b$ if the expression is bounded.
