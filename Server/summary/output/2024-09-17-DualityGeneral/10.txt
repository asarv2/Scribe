\textbf{Primal Problem (P)}
\begin{itemize}
    \item $\max \xi(x) = f_0(x)$
    \item s.t. $f_i(x) \le 0$, $i=1,2,\dots,m$
    \item $g_j(x) = 0$, $j=1,2,\dots,n$
\end{itemize}

\textbf{Dual Problem (D)}
\begin{itemize}
    \item $\min \xi(\lambda,\mu) = \max_{x \in \mathbb{R}^n} [f_0(x) - \sum_{i=1}^m \lambda_i f_i(x) - \sum_{j=1}^n \mu_j g_j(x)]$
    \item s.t. $\lambda_i \ge 0$, $i=1,2,\dots,m$
\end{itemize}

\textbf{Weak Duality}
$\xi^* = \max_{x \in \mathcal{G}} \xi(x) \le \min_{\lambda \ge 0, \mu} \xi(\lambda,\mu) = \xi^*$

The dual problem (D) is to minimize the function $\xi(\lambda,\mu)$, which is the maximum of the Lagrangian function over all possible values of $x$, subject to the non-negativity constraint $\lambda_i \ge 0$ for $i = 1, \dots, m$.
