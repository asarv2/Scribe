\section*{Basic and Non-Basic Variables}

$\mathbf{x} = \begin{pmatrix}
x_1 \\
\vdots \\
x_n \\
w_1 \\
\vdots \\
w_m
\end{pmatrix} = \begin{pmatrix}
\mathbf{x}_B \\
\mathbf{x}_N
\end{pmatrix}$

$\mathbf{A} \mathbf{x} + \mathbf{w} = \mathbf{b} \iff \sum_{j=1}^n a_{ij} x_j + x_{n+i} = b_i$

$\begin{bmatrix} \mathbf{A} & \mathbf{I} \end{bmatrix} \begin{bmatrix} \mathbf{x} \\ \mathbf{w} \end{bmatrix} = \mathbf{b}$

$\sum_{j \in B} a_{ij} x_j + \sum_{j \in N} a_{ij} x_j + x_{n+i} = b_i$

$\mathbf{x}_B + \mathbf{x}_N = \mathbf{b}$
