\section*{Dual Simplex Method  \textit{LVJ p.71}}

We begin with an example:

\textbf{Step 0}
\begin{itemize}
    \item maximize $-x_1 - x_2$
    \item subject to
        \begin{itemize}
            \item $-2x_1 + x_2 \le 4$
            \item $-2x_1 + 4x_2 \le -8$
            \item $-x_1 + 3x_2 \le -7$
            \item $x_1, x_2 \ge 0$
        \end{itemize}
\end{itemize}
\textbf{The dual of this problem is}
\begin{itemize}
    \item minimize $4y_1 - 8y_2 - 7y_3$
    \item subject to
        \begin{itemize}
            \item $-2y_1 - 2y_2 - y_3 \ge -1$
            \item $-y_1 + 4y_2 + 3y_3 \ge -1$
            \item $y_1, y_2, y_3 \ge 0$
        \end{itemize}
\end{itemize}

\end{document}

```latex
    \item $\mathbf{x}_B = \mathbf{x}_B - (\mathbf{B}^{-1} \mathbf{N}) \mathbf{x}_N$
\end{itemize}

\begin{itemize}
    \item $\xi^* = \mathbf{c}_B^T \mathbf{B}^{-1} \mathbf{b}$, $\mathbf{z}_N = (\mathbf{B}^{-1} \mathbf{N})^T \mathbf{c}_B - \mathbf{c}_N$, $\mathbf{x}_B = \mathbf{B}^{-1} \mathbf{b}$
\end{itemize}

\end{document}
```
