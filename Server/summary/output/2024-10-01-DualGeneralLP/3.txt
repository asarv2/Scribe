Slide 3

\begin{align*}
\vec{y}^T \vec{A} (\vec{x}^+ - \vec{x}^-) &\le \vec{y}^T \vec{b} + \vec{z}^T \vec{u} \\
\vec{z}^T (\vec{x}^+ - \vec{x}^-) &\le \vec{z}^T \vec{u} \\
(\vec{y}^T \vec{A} + \vec{z}^T) \vec{x}^+ + (-\vec{y}^T \vec{A} - \vec{z}^T) \vec{x}^- &\le \vec{y}^T \vec{b} + \vec{z}^T \vec{u} \\
\implies \vec{c}^T &\le (\vec{y}^T \vec{A} + \vec{z}^T) \\
\end{align*}

\newline
\textbf{(D)}
\begin{align*}
\min \ \vec{b}^T \vec{y} + \vec{u}^T \vec{z} \\
\text{s.t.} \quad \vec{A}^T \vec{y} + \vec{z} &\ge \vec{c} \\
\qquad \vec{y}, \vec{z} &\ge \vec{0}
\end{align*}
