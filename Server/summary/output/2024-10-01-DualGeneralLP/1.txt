\textbf{Dual of General LP Problem}

\newline
\textit{(EVJ p.107 / #66, Chapter 9, p 152)}

\newline
\textbf{Recall (P)}
\begin{align*}
\max \ \vec{s} &= \vec{c} \vec{x} \\
\text{s.t.} \quad \vec{A} \vec{x} &\le \vec{b} \\
\qquad \vec{x} &\ge \vec{0}
\end{align*}

\newline
\textbf{(D)}
\begin{align*}
\min \ \vec{s} &= \vec{b} \vec{y} \\
\text{s.t.} \quad \vec{A}^T \vec{y} &\ge \vec{c} \\
\qquad \vec{y} &\ge \vec{0}
\end{align*}

\newline
\textbf{Derivation:}
Take linear combination of the constraints.
\newline
$y$ - Lagrange Multiplier

\newline
$(\vec{y}^T \vec{A}) \vec{x} \le \vec{y}^T \vec{b} \implies (\vec{y}^T \vec{A}) \vec{x} \le \vec{y}^T \vec{b}$

\newline
$\vec{c}^T \vec{x} \le (\vec{y}^T \vec{A}) \vec{x} \le \vec{y}^T \vec{b}$
