Slide 4

\textbf{M2}
\begin{align*}
\max \ \vec{c}^T \vec{x} \\
\text{s.t.} \quad \vec{A} \vec{x} \le \vec{b} \\
\qquad \vec{x} \le \vec{u}
\end{align*}

\newline
$\max \ \mathcal{S}(\vec{x}) = \vec{c}^T \vec{x}$
\newline
\textit{s.t.}
\begin{align*}
\vec{A} \vec{x} - \vec{b} &\le \vec{0} \\
\vec{x} - \vec{u} &\le \vec{0}
\end{align*}

\newline
\textbf{Lagrangian function}
\newline
$\mathcal{L}(\vec{x}, \vec{y}, \vec{z}) = \vec{c}^T \vec{x} - \vec{y}^T (\vec{A} \vec{x} - \vec{b}) - \vec{z}^T (\vec{x} - \vec{u})$
\newline
$\vec{y} \ge \vec{0}$
\newline
$\vec{z} \ge \vec{0}$
