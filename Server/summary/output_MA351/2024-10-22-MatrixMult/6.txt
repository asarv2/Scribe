\section*{Properties of Matrix Multiplication}

\begin{enumerate}
    \item $A(\alpha B + \beta C) = \alpha AB + \beta AC$
    \item $(\alpha A + \beta B) C = \alpha AC + \beta BC$
    \item $(AB)^T = B^T A^T$
    \item In general, $AB \neq BA$
\end{enumerate}
