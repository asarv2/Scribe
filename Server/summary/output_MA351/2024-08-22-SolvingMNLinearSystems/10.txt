\begin{equation*}
\begin{cases}
x + y - z = 2 \\
3x - y + 2z = 0 \\

\end{cases}
\end{equation*}

\begin{equation*}
\begin{pmatrix}

3 & -1 & 2 \\
2 & -2 & 3
\end{pmatrix}
\rightarrow
\begin{pmatrix}
1 & 1 & -1 \\

0 & -4 & 5
\end{pmatrix}
\rightarrow
\begin{pmatrix}

\end{pmatrix}
\end{equation*}
