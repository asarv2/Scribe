\section*{Properties of Vector Add. \& Scalar Mult.}

$(\vec{u} + \vec{v} \in R^n, \quad \alpha \vec{u} \in R^n)$

\begin{enumerate}
    \setcounter{enumi}{5}
    \item Let $\vec{0} = \begin{pmatrix} 0 \\ 0 \\ \vdots \\ 0 \end{pmatrix}$. Then $\vec{0} + \vec{u} = \vec{u}$
    \item Let $-\vec{u} = \begin{pmatrix} -x_1 \\ -x_2 \\ \vdots \\ -x_n \end{pmatrix}$. Then $(-\vec{u}) + \vec{u} = \vec{0}$
    \item $1\vec{u} = \vec{u}$
\end{enumerate}
