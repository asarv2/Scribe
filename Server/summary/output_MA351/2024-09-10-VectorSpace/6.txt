\section*{Linear Combination and Span}

\noindent
\textbf{8.} Given $\mathcal{L} = \{\vec{u}_1, \vec{u}_2, ..., \vec{u}_k\} \subseteq R^n$.
\newline
\noindent
Span$\{\vec{u}_1, \vec{u}_2, ..., \vec{u}_k\}$ (or Simply Span($\mathcal{L}$))
\newline
\noindent
= \{the collection of all the possible lin. comb. of the $\vec{u}_1, \vec{u}_2, ..., \vec{u}_k\}$
\newline
\noindent
= $\{\vec{v} : \vec{v} = c_1 \vec{u}_1 + c_2 \vec{u}_2 + ... + c_k \vec{u}_k\}$
