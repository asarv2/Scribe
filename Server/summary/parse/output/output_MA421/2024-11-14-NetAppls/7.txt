\section*{Inequality constraints (More generally)}
\textbf{[C] p. 322}

$[A_1]$ - incident matrix
$[A_2]$
\begin{itemize}
    \item $A_1 \bar{x} \ge -\bar{b}_1$
    \item $A_2 \bar{x} \le -\bar{b}_2$
    \item $A_1 \bar{x} = -\bar{b}_1 + \bar{w}_1$, $\bar{w}_1 \ge 0$
    \item $A_2 \bar{x} = -\bar{b}_2 - \bar{w}_2$, $\bar{w}_2 \ge 0$
\end{itemize}

new constraint:
$\sum_{p} w_{1p} - \sum_{s} w_{s2} = \sum_{p} b_{1p} + \sum_{s} b_{s2}$
