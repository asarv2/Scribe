\textbf{[C] p.67}

\begin{equation*}
\max. \ \zeta = 40x_1 + 70x_2
\end{equation*}
\begin{equation*}
\text{(P)} \quad \text{s.t.} \begin{cases}
x_1 + x_2 \le 100 \\
x_1 + 30x_2 \le 4000 \\
x_1, x_2 \ge 0
\end{cases}
\end{equation*}

\textbf{Opt. Solution:} $x_1^* = 25$, $x_2^* = 75$, $\zeta^* = 40(25) + 70(75) = 6250$

\textbf{Q.}
\begin{enumerate}
    \item Check optimality.
    \item What if 4000 is changed to 4000+t? How would $x_1^*$, $x_2^*$, $\zeta^*$ be changed? How large can t be?
\end{enumerate}
