\section*{Theorem 10.1}
$S$ is convex $\iff$
it contains all conv. comb. of points in $S$

\textbf{Proof} $\Rightarrow$
Suppose $S$ is convex.
$n=2$: $z_1, z_2 \in S \implies \lambda z_1 + (1-\lambda) z_2 \in S$
$0 \le \lambda \le 1$

$n=3$: $z_1, z_2, z_3 \in S \implies \lambda_1 z_1 + \lambda_2 z_2 + \lambda_3 z_3 \in S$
$\lambda_1 + \lambda_2 + \lambda_3 = 1$
$\lambda_1, \lambda_2, \lambda_3 \ge 0$

$z_3 \in S \implies \lambda_1 z_1 + \lambda_2 z_2 + \lambda_3 z_3 = (\lambda_1 + \lambda_2) \left(\frac{\lambda_1 z_1 + \lambda_2 z_2}{\lambda_1 + \lambda_2}\right) + \lambda_3 z_3 \in S$
$\lambda_1 + \lambda_2 \le 1$
$\lambda_1, \lambda_2 \ge 0$
$\lambda_3 \ge 0$
