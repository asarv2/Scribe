\begin{equation*}
\text{Non-Singular Matrix } A_{mxn} \text{ is called a square matrix if } m = n
\end{equation*}
\begin{equation*}
\text{Given } A_{mxn} \text{ and } B_{m \times 1}
\end{equation*}
\begin{equation*}
\text{If } AX = B \text{ has a unique solution, then } A \text{ is non-singular}
\end{equation*}
\begin{equation*}
\text{If } AX = B \text{ has no solution, then } A \text{ is singular}
\end{equation*}
\begin{equation*}
\text{If } AX = B \text{ has infinitely many solutions, then } A \text{ is singular}
\end{equation*}
\begin{equation*}
\text{If } AX = B \text{ has a unique solution, then } A \text{ is non-singular}
\end{equation*}
\begin{equation*}
\text{If } A \text{ is non-singular, then } AX = B \text{ has a unique solution for any } B
\end{equation*}
\begin{equation*}
\text{If } A \text{ is a square matrix, then } A \text{ is non-singular} \iff \text{Rank(A)} = n
\end{equation*}
