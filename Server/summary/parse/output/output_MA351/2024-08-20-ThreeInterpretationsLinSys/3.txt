\textbf{(II)} Writing vector as linear combination
(look at "columns")

\begin{align*}
\begin{pmatrix}
2x - y \\
x + 3y
\end{pmatrix} &= \begin{pmatrix}
4 \\
9
\end{pmatrix}
\end{align*}
\textit{vector addition}
\begin{align*}
\begin{pmatrix}
2x \\
x
\end{pmatrix} + \begin{pmatrix}
-y \\
3y
\end{pmatrix} &= \begin{pmatrix}
4 \\
9
\end{pmatrix}
\end{align*}
\textit{scalar multiplication}
\begin{align*}
x\begin{pmatrix}
2 \\
1
\end{pmatrix} + y\begin{pmatrix}
-1 \\
3
\end{pmatrix} &= \begin{pmatrix}
4 \\
9
\end{pmatrix}
\end{align*}
