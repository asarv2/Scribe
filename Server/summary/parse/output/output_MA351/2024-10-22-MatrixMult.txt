$A_{mxn} \cdot X \in R^n \rightarrow AX \in R^m$

\vdots \\

\end{pmatrix}
\end{equation*}

\textit{"Formula"}

\begin{enumerate}

    \item $(\alpha A + \beta B) X = \alpha AX + \beta BX$
\end{enumerate}

\section*{Properties of Matrix Multiplication}

\begin{enumerate}
    \item $A(\alpha B + \beta C) = \alpha AB + \beta AC$
    \item $(\alpha A + \beta B) C = \alpha AC + \beta BC$
    \item $(AB)^T = B^T A^T$
    \item In general, $AB \neq BA$
\end{enumerate}
