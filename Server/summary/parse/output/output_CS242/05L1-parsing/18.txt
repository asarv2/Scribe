links = soup.find_all('a')
# get text associated with links
for link in links:
    linkText = link.get_text().strip()
    print(linkText)
# get urls
for link in links:
    # gets the href attribute
    linkUrl = link["href"]
    # filter for html pages with absolute path
    if 'http' in linkUrl and '.html' in linkUrl:
        print(linkUrl)
