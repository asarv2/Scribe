```python
def mapreduce_execute(data, mapper, reducer):
    values = map(mapper, data)
    groups = {}
    for items in values:
        for k, v in items:
            if k not in groups:
                groups[k] = [v]
            else:
                groups[k].append(v)
    output = [reducer(k, v) for k, v in groups.items()]
    return output
```
