categories = iris.category.unique()
c = ["#00A5FC","#007979","#008800","#215419"]
byCategory = {x:iris[iris.category==x] for x in categories}
swMean = [byCategory[y]['sepal-width'].mean() for y in byCategory.keys()]
slMean = [byCategory[y]['sepal-length'].mean() for y in byCategory.keys()]
pwMean = [byCategory[y]['petal-width'].mean() for y in byCategory.keys()]
plMean = [byCategory[y]['petal-length'].mean() for y in byCategory.keys()]
