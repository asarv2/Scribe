```
Anscombe colors and marks
import pandas as pd
import matplotlib.pyplot as plt
%matplotlib inline
# This will also make it easier to iterate
data = pd.read_csv('anscombe.csv',sep=',')
X = data[['x1','x2','x3','x4']]
Y = data[['y1','y2','y3','y4']]
mark = ["o","*","X","^"]
colours = ["#00A5FC","#008800","#007979","#215419"]
for i in range(0,4):
    plt.scatter(X.iloc[:,i],Y.iloc[:,i], marker = mark[i],
                label="Set "+str(i+1), c=colours[i])
plt.legend(loc='best')
```
