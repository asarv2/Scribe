```python
def pow(x, n=1):
    """x to the nth power"""  # documentation string, for help()
    if n <= 0:
        return 1
    else:
        half = pow(x, n // 2)
        two = half * half
        if (n % 2 is 1):
            return x * two  # return indicates the value to be sent back
        return two
```
