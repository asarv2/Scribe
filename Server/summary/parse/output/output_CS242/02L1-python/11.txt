```markdown
# Playing with Python

>>> x=['a','b','c']
>>> x[1]
'b'
>>> x[1:]
['b', 'c']
>>> x[1]='d'
>>> x
['a', 'd', 'c']
>>> d={'alice':1234, 'bob':5678, 'clare':9012}
>>> d['bob']
5678
>>> d['bob'] = 7777
>>> d
{'clare': 9012, 'bob': 7777, 'alice': 1234}
>>> d.keys()
dict.keys(['clare', 'bob', 'alice'])
>>> d.items()
dict.items([('clare', 9012), ('bob', 7777), ('alice', 1234)])
>>> del d['bob']
```
