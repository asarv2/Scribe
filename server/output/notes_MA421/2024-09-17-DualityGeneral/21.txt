\textbf{Strong Duality for "Traditional" LP}

\begin{itemize}
    \item \textbf{Primal Problem (P)}
    \begin{align*}
        \max \xi(x) &= c^T x \\
        \text{s.t.} \quad Ax &\le b \\
        x &\ge 0
    \end{align*}
    \item \textbf{Dual Problem (D)}
    \begin{align*}
        \min \xi(\lambda) &= \lambda^T b \\
        \text{s.t.} \quad A^T \lambda &\ge c \\
        \lambda &\ge 0
    \end{align*}
    \item \textbf{Strong Duality Theorem}
    \begin{itemize}
        \item If the primal problem (P) has a finite optimal solution, then the dual problem (D) also has a finite optimal solution, and the optimal values of the primal and dual problems are equal.
    \end{itemize}
\end{itemize}
