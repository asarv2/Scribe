\textbf{Primal and Dual Problems}

\begin{itemize}
    \item[(P)]
    \begin{align*}
        \max \ \sum_{j} c_j x_j &= c^T x \\
        \text{subject to} \quad \sum_{j} a_{ij} x_j &\le b_i, \quad i = 1, \dots, m \\
        x_j &\ge 0, \quad j = 1, \dots, n
    \end{align*}
    \item[(D)]
    \begin{align*}
        \min \ \sum_{i} b_i y_i &= b^T y \\
        \text{subject to} \quad \sum_{i} a_{ij} y_i &\ge c_j, \quad j = 1, \dots, n \\
        y_i &\ge 0, \quad i = 1, \dots, m
    \end{align*}
\end{itemize}

\textbf{Theorem 5.1: Weak Duality}
\begin{center}
    $\sum_{j} c_j x_j \le \sum_{i} b_i y_i$
\end{center}
